\node[etapa=coral, text=black, right=of t1, xshift=19mm] (t2) {Annotation};
\node[cc, below=3.5mm of t2] (a1) {1. Seed terms (CASS glossary, 38)};

\node[modelo, draw=colcrf, fill=colcrf!12, below=of m1] (m2) {CRF\\ (27 features + sequence)};
\node[modelo, draw=colbil, fill=colbil!12, below=of m2] (m3) {BiLSTM-CRF\\ (embeddings learnt from scratch)};
\node[modelo, draw=colxlm, fill=colxlm!12, below=of m3] (m4) {XLM-RoBERTa\\ (pretrained, fine-tuned)};
